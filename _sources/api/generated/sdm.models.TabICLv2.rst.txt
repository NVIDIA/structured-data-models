Default Recipe
--------------




.. code-block:: python

   Recipe(
     features=Sequential(
       StypeDispatch(
         categorical=Sequential(
           AlignCategories(sort_by='value'),
           ToNumerical(),
         ),
       ),
       StypeDispatch(
         numerical=Sequential(
           ImputeMean(),
           DropConstantColumns(),
           Standardize(eps=1e-06),
           Clip(-100.0, 100.0),
           Choice(
             Identity(),
             PowerTransform(),
             method='round_robin',
           ),
           ClipSigma(threshold=4.0, method='log'),
           ShuffleColumns(method='latin'),
         ),
       ),
     ),
     target=Sequential(
       StypeDispatch(
         numerical=Standardize(),
         categorical=Sequential(
           AlignCategories(sort_by='value'),
           ShuffleCategories(method='shift'),
         ),
       ),
     ),
     output=Sequential(
       TaskDispatch(
         regression=SortQuantiles(),
       ),
       AverageEstimators(),
       TaskDispatch(
         classification=Softmax(temperature=0.9),
       ),
     ),
   )
