Default Recipe
--------------

.. code-block:: python

   Recipe(
     features=Sequential(
       StypeDispatch(
         categorical=Sequential(
           AlignCategories(sort_by='value'),
           ToNumerical(),
         ),
       ),
       StypeDispatch(
         numerical=Sequential(
           ImputeMean(),
           DropConstantColumns(),
           Standardize(),
           Clip(-100.0, 100.0),
           Choice(
             Identity(),
             PowerTransform(),
             method='round_robin',
           ),
           ClipSigma(threshold=4.0),
           ShuffleColumns(method='shift'),
         ),
       ),
     ),
     target=Sequential(
       StypeDispatch(
         numerical=Standardize(),
         categorical=Sequential(
           AlignCategories(),
           ShuffleCategories(method='shift'),
         ),
       ),
     ),
     output=Sequential(
       ReduceEstimators(method='mean'),
       TaskDispatch(
         classification=Softmax(temperature=0.9),
       ),
     ),
   )
