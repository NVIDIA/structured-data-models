Default Recipe
--------------

.. code-block:: python

   Recipe(
     features=Sequential(
       StypeDispatch(
         categorical=Sequential(
           AlignCategories(sort_by='value', min_frequency=2),
           ToNumerical(),
         ),
       ),
       StypeDispatch(
         numerical=Sequential(
           DropConstantColumns(),
           Standardize(eps=1e-06),
           Clip(-100.0, 100.0),
           Choice(
             Identity(),
             PowerTransform(),
             method='round_robin',
           ),
           ClipSigma(threshold=4.0),
           FlipSign(),
           ShuffleColumns(method='latin'),
           SelectColumns(),
         ),
       ),
     ),
     target=Sequential(
       StypeDispatch(
         numerical=Sequential(
           Standardize(eps=0.0),
           FlipSign(),
         ),
         categorical=Sequential(
           AlignCategories(),
           ShuffleCategories(method='shift'),
         ),
       ),
     ),
     output=Sequential(
       TaskDispatch(
         regression=SortQuantiles(),
       ),
       ReduceEstimators(method='mean'),
       TaskDispatch(
         classification=Softmax(),
       ),
     ),
   )
