Default Recipe
--------------

.. code-block:: python

   Recipe(
     features=Sequential(
       StypeDispatch(
         numerical=Sequential(
           Cast(torch.float64),
           DropConstantColumns(),
           Standardize(eps=1e-06),
           Clip(-100.0, 100.0),
           Choice(
             Identity(),
             PowerTransform(),
             Sequential(
               RobustScale(quantile_range=(25.0, 75.0)),
               ClipSoft(3.0),
             ),
             method='round_robin',
           ),
           ClipSigma(threshold=4.0),
           FlipSign(probability=0.5),
         ),
         categorical=Sequential(
           AlignCategories(sort_by='value'),
           AddCategoryCounts(min_cardinality=50),
           ToNumerical(),
           Cast(torch.float64),
           DropConstantColumns(),
           Standardize(eps=1e-06),
           Clip(-100.0, 100.0),
           Choice(
             Identity(),
             PowerTransform(),
             Sequential(
               RobustScale(quantile_range=(25.0, 75.0)),
               ClipSoft(3.0),
             ),
             method='round_robin',
           ),
           ClipSigma(threshold=4.0),
         ),
       ),
       ShuffleColumns(method='latin'),
       SelectColumns(),
       Cast(torch.float32),
     ),
     target=Sequential(
       StypeDispatch(
         numerical=Sequential(
           Standardize(eps=0.0),
           FlipSign(probability=0.5),
         ),
         categorical=Sequential(
           AlignCategories(),
           ShuffleCategories(method='shift'),
         ),
       ),
     ),
     output=TaskDispatch(
       classification=Sequential(
         AverageEstimators(),
         Softmax(),
       ),
       regression=Sequential(
         SortQuantiles(),
         AverageEstimators(trim_fraction=0.2),
       ),
     ),
   )
